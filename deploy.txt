# 前準備
az webapp stop --name hirahira-room --resource-group hirahira

# デプロイ
＃az group delete --name hirahira --yes　（全リソース削除のため実行しないこと）
az webapp delete --name hirahira-room --resource-group hirahira
az webapp up --name hirahira-room --runtime "PYTHON|3.11" --resource-group hirahira --location japanwest

# ポータル画面でVNETを追加する。

# データベース（PostgreSQL サーバー hirahira-db 上の専用DB hirahira_room を使う）
# 同じサーバーに funitclub 用の DB もあるため、既定の postgres DB は使わない。
# サーバーは公開アクセス無効だが、DB の作成は管理用の経路なので手元から実行できる。
az postgres flexible-server db create --resource-group hirahira --server-name hirahira-db --database-name hirahira_room --charset UTF8 --collation en_US.utf8
az webapp config appsettings set --name hirahira-room --resource-group hirahira --settings DB_NAME=hirahira_room

# 起動プローブの待ち時間を延ばす（既定230秒。B1では起動が間に合わずコンテナが落とされる）
az webapp config appsettings set --name hirahira-room --resource-group hirahira --settings WEBSITES_CONTAINER_START_TIME_LIMIT=600

# マイグレーション（本番＝Azure 上で実行する）
# Oryx の圧縮ビルドのため、アプリの実体は /tmp/<ID>/ に展開されている。
# ID は起動ごとに変わるので探して移動し、同梱の antenv を有効にしてから実行する。
az webapp ssh --name hirahira-room --resource-group hirahira

# 再起動
az webapp restart --name hirahira-room --resource-group hirahira
az webapp start --name hirahira-room --resource-group hirahira
az webapp show --name hirahira-room --resource-group hirahira --query state
